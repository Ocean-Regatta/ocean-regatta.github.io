let cachedData = defaultLeaderboard;

function renderTable(teams) {
  const tbody = document.getElementById("scoreboard-body");
  if (!tbody) return;
  tbody.innerHTML = "";

  teams.forEach(t => {
    const tr = document.createElement("tr");
    tr.style.borderBottom = "1px solid var(--md-default-fg-color--lightest, #334155)";
    
    // Rank medal
    let rankBadge = `<span style="font-weight:bold;">#${t.rank}</span>`;
    if (t.rank === 1) rankBadge = `<span style="color:#fbbf24; font-weight:bold; font-size:1.05rem;">🥇 #1</span>`;
    else if (t.rank === 2) rankBadge = `<span style="color:#94a3b8; font-weight:bold; font-size:1.05rem;">🥈 #2</span>`;
    else if (t.rank === 3) rankBadge = `<span style="color:#cd7f32; font-weight:bold; font-size:1.05rem;">🥉 #3</span>`;

    // Status badge
    let statusColor = "#10b981";
    let statusBg = "rgba(16,185,129,0.15)";
    let statusText = t.status;
    if (t.status === "COLLISION") {
      statusColor = "#ef4444";
      statusBg = "rgba(239,68,68,0.15)";
    } else if (t.status === "TIMEOUT") {
      statusColor = "#f59e0b";
      statusBg = "rgba(245,158,11,0.15)";
    }

    // Mini badge icons next to team name
    const miniBadges = (t.badges || []).map(b => 
      `<span style="margin-left:3px; cursor:help; font-size:0.95rem;" title="${b.name}: ${b.desc}">${b.icon}</span>`
    ).join("");

    // Badges flex tags that wrap inside cell
    const badgeTags = (t.badges || []).map(b => {
      const isExcl = b.exclusive;
      const borderCol = isExcl ? "#f59e0b" : "#0284c7";
      const bgCol = isExcl ? "rgba(245,158,11,0.12)" : "rgba(2,132,199,0.12)";
      return `<span style="display:inline-flex; align-items:center; gap:2px; padding:1px 5px; border-radius:8px; font-size:0.7rem; line-height:1.2; border:1px solid ${borderCol}; background:${bgCol}; cursor:help; white-space:nowrap;" title="${b.desc}">${b.icon} ${b.name}</span>`;
    }).join("");

    tr.innerHTML = `
      <td style="padding: 0.5rem 0.3rem; text-align:center;">${rankBadge}</td>
      <td style="padding: 0.5rem 0.5rem; font-weight:600;">
        <span style="color:var(--md-primary-fg-color, #38bdf8);">${t.display_name || t.team}</span>
        <span style="margin-left:3px;">${miniBadges}</span>
      </td>
      <td style="padding: 0.5rem 0.3rem; text-align:center; font-weight:bold; color:#10b981;">${t.score} pts</td>
      <td style="padding: 0.5rem 0.3rem; text-align:center; font-family:monospace; font-size:0.8rem;">${t.sim_time ? t.sim_time.toFixed(1) + 's' : '-'}</td>
      <td style="padding: 0.5rem 0.3rem; text-align:center;">
        <span style="display:inline-block; padding:1px 5px; border-radius:4px; font-size:0.7rem; font-weight:600; color:${statusColor}; background:${statusBg}; white-space:nowrap;">${statusText} (${t.waypoints || '-'})</span>
      </td>
      <td style="padding: 0.5rem 0.5rem;">
        <div style="display:flex; flex-wrap:wrap; gap:3px; align-items:center;">
          ${badgeTags || '<span style="color:var(--md-default-fg-color--lightest); font-size:0.75rem;">-</span>'}
        </div>
      </td>
    `;
    tbody.appendChild(tr);
  });
}

function filterScoreboard() {
  const query = (document.getElementById("scoreboard-search")?.value || "").toLowerCase().trim();
  if (!query) {
    renderTable(cachedData);
    return;
  }
  const filtered = cachedData.filter(t => {
    const nameMatch = (t.display_name || t.team).toLowerCase().includes(query);
    const badgeMatch = (t.badges || []).some(b => b.name.toLowerCase().includes(query));
    const statusMatch = (t.status || "").toLowerCase().includes(query);
    return nameMatch || badgeMatch || statusMatch;
  });
  renderTable(filtered);
}

const candidatePaths = [
  "../../data/leaderboard_2026.json",
  "../../data/leaderboard.json",
  "../data/leaderboard.json",
  "data/leaderboard.json"
];

function fetchLeaderboard(paths) {
  if (!paths || paths.length === 0) {
    renderTable(defaultLeaderboard);
    return;
  }
  fetch(paths[0])
    .then(res => {
      if (!res.ok) throw new Error("Fetch failed for " + paths[0]);
      return res.json();
    })
    .then(json => {
      if (json && json.teams && json.teams.length > 0) {
        cachedData = json.teams;
        const leader = json.teams[0];
        if (leader) {
          document.getElementById("leader-name").innerText = leader.display_name || leader.team;
          document.getElementById("leader-score").innerText = leader.score + " pts";
          document.getElementById("leader-time").innerText = leader.sim_time.toFixed(1) + " s";
        }
        renderTable(cachedData);
      } else {
        renderTable(defaultLeaderboard);
      }
    })
    .catch(() => {
      fetchLeaderboard(paths.slice(1));
    });
}

fetchLeaderboard(candidatePaths);
